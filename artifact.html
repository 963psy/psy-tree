let clientAnswers = JSON.parse(localStorage.getItem('psychTree_answers')) || {}; let completedNodes = new Set(JSON.parse(localStorage.getItem('psychTree_completed')) || []); let selectedKey = null; // به‌روزرسانی وضعیت در شروع window.onload = () => { completedNodes.forEach(key => { const spot = document.getElementById(`spot-${key}`); if (spot) spot.classList.add('completed'); }); updateProgress(); }; function openNode(key) { selectedKey = key; const node = clinicalData[key]; document.getElementById('card-title').innerText = node.title; document.getElementById('card').style.display = 'block'; document.getElementById('client-response').value = clientAnswers[key] || ''; changeTab('concept'); } function closeCard() { document.getElementById('card').style.display = 'none'; } function changeTab(tabName) { document.getElementById('tab-concept').classList.toggle('active', tabName === 'concept'); document.getElementById('tab-scenario').classList.toggle('active', tabName === 'scenario'); document.getElementById('tab-exercise').classList.toggle('active', tabName === 'exercise'); if (selectedKey && clinicalData[selectedKey]) { document.getElementById('card-body').innerText = clinicalData[selectedKey][tabName]; } const inputBox = document.getElementById('exercise-input-box'); inputBox.style.display = (tabName === 'exercise') ? 'block' : 'none'; } function finishNode() { const text = document.getElementById('client-response').value.trim(); if (text) { clientAnswers[selectedKey] = text; localStorage.setItem('psychTree_answers', JSON.stringify(clientAnswers)); } completedNodes.add(selectedKey); localStorage.setItem('psychTree_completed', JSON.stringify(Array.from(completedNodes))); document.getElementById(`spot-${selectedKey}`).classList.add('completed'); updateProgress(); closeCard(); } function updateProgress() { const total = Object.keys(clinicalData).length; const count = completedNodes.size; const percent = Math.round((count / total) * 100); document.getElementById('progress-fill').style.width = percent + '%'; document.getElementById('progress-text').innerText = percent + '%'; if (count >= 1) { document.getElementById('btn-show-report').style.display = 'block'; } } function showReport() { const container = document.getElementById('report-body'); container.innerHTML = ''; for (let key in clinicalData) { const item = clinicalData[key]; const answer = clientAnswers[key] || "هنوز پاسخی ثبت نشده است."; const box = document.createElement('div'); box.className = 'report-item'; box.innerHTML = `

${item.title}

تمرین بالینی: ${item.exercise}

پاسخ مراجع: ${answer}

`; container.appendChild(box); } document.getElementById('report-modal').style.display = 'flex'; } function closeReport() { document.getElementById('report-modal').style.display = 'none'; }
